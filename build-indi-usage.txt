# Install everything (INDI + LibXISF + all 3rd party) - DEFAULT
./build-indi-stable.sh

# Install INDI + stellarsolver + KStars
./build-indi-stable.sh --stellasolver --kstars

# Install INDI + PHD2 with specific version
./build-indi-stable.sh --phd2 v2.6.11

# Install INDI + KStars + PHD2
./build-indi-stable.sh --kstars --phd2

# Install only specific drivers + stellarsolver
./build-indi-stable.sh --3rdparty my_drivers.txt --stellasolver

# Install everything except 3rd party drivers
./build-indi-stable.sh --no-3rdparty --stellasolver --kstars --phd2
